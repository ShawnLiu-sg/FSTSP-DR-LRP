cmake_minimum_required(VERSION 3.16)
project(fstsp_dr_lrp LANGUAGES CXX)

set(GUROBI_ROOT "$ENV{GUROBI_HOME}" CACHE PATH "Gurobi C/C++ SDK root (win64 on Windows)")
find_path(GUROBI_INCLUDE gurobi_c++.h HINTS "${GUROBI_ROOT}/include")
find_library(GUROBI_C NAMES gurobi120 HINTS "${GUROBI_ROOT}/lib")
if(MSVC)
    find_library(GUROBI_CXX NAMES gurobi_c++md2017 HINTS "${GUROBI_ROOT}/lib")
    find_library(GUROBI_CXX_DEBUG NAMES gurobi_c++mdd2017 HINTS "${GUROBI_ROOT}/lib")
    if(NOT GUROBI_CXX_DEBUG)
        message(FATAL_ERROR "Cannot find gurobi_c++mdd2017. Set GUROBI_ROOT to the Gurobi 12 SDK.")
    endif()
else()
    find_library(GUROBI_CXX NAMES gurobi_c++ HINTS "${GUROBI_ROOT}/lib")
endif()
if(NOT GUROBI_INCLUDE OR NOT GUROBI_C OR NOT GUROBI_CXX)
    message(FATAL_ERROR "Set GUROBI_ROOT to the Gurobi 12 SDK directory with include and lib.")
endif()

add_executable(fstsp_dr_lrp src/main.cpp src/instance.cpp src/timing.cpp src/sdrp.cpp src/search.cpp)
target_compile_features(fstsp_dr_lrp PRIVATE cxx_std_17)
target_include_directories(fstsp_dr_lrp PRIVATE "${GUROBI_INCLUDE}")
if(MSVC)
    target_compile_options(fstsp_dr_lrp PRIVATE /W4 /permissive- /utf-8)
    target_link_libraries(fstsp_dr_lrp PRIVATE optimized "${GUROBI_CXX}" debug "${GUROBI_CXX_DEBUG}" "${GUROBI_C}")
else()
    target_compile_options(fstsp_dr_lrp PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(fstsp_dr_lrp PRIVATE "${GUROBI_CXX}" "${GUROBI_C}")
endif()
